from guppylang import guppy
from guppylang.std.quantum import qubit, measure, h, z, ry, rz, cx, t, angle
from guppylang.std.builtins import output

@guppy
def main() -> None:
    # 4 qubits Dancer A, 4 qubits Dancer B, 2 Hype Referee Ancillae
    q0, q1, q2, q3 = qubit(), qubit(), qubit(), qubit()
    q4, q5, q6, q7 = qubit(), qubit(), qubit(), qubit()
    q8, q9 = qubit(), qubit()

    h(q0); h(q4)
    ry(q1, angle(0.785398)); ry(q5, angle(0.785398))
    cx(q0, q1); cx(q4, q5)
    rz(q2, angle(1.570796)); rz(q6, angle(1.570796))
    cx(q1, q2); cx(q5, q6); cx(q2, q3); cx(q6, q7)

    # Mid-circuit measurement of Hype Referee ancilla q8
    h(q8); cx(q3, q8); cx(q7, q8); t(q8)
    hype_active = measure(q8).read()

    if hype_active:
        z(q2); z(q6)           # Thoracic CHEST POP phase kick
        cx(q0, q4); cx(q1, q5) # Cross-dancer Bell lock

    output("hype", hype_active)
    output("q0", measure(q0).read())
